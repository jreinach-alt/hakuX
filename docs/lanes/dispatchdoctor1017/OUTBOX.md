```bash
import sys
since = int(sys.argv[1]); D = sys.argv[2]
rows = []
try:
    with open(D + '/logs/dispatch-doctor.tsv') as f:

            parts = line.rstrip('\n').split('\t')
            if len(parts) < 4:
                continue
            try:
                t0 = int(parts[0])
            except ValueError:
                continue
            if t0 >= since:
                rows.append(parts)
except FileNotFoundError:

if rows is None:
    print("  dispatch doctor: no sessions yet (logs/dispatch-doctor.tsv does not exist)")
else:
    n = len(rows)
    fixed = sum(1 for r in rows if r[3] == 'retry')
    gaveup = n - fixed
    print("  dispatch doctor: %d sessions, %d fixed, %d gave up" % (n, fixed, gaveup))
PYDOCTOR
```
